+ a/personas/g1.grok.toml  f 600
    | # /etc/grok/managed_config.toml in grok persona g1, written by nexus --config from personas/grok.toml
    | # (grok settings for every grok persona here) and, with the statusline on, statusline/grok/settings.toml.
    | # Edit those and re-run nexus --config, not this file. A key in the persona's ~/.grok/config.toml wins.
    | 
    | # personas/grok.toml: grok settings for every grok persona, on every VM that pulls this repo. nexus --config
    | # copies it into each grok persona's managed config (personas/<name>.grok.toml, /etc/grok/managed_config.toml
    | # inside, with the statusline after it) and checks it first, so re-run nexus --config after an edit: the next grok
    | # started in a persona reads it.
    | #
    | # It is grok's own config.toml format (~/.grok/docs/user-guide/26-config-reference.md in a persona). These
    | # are defaults: a key in a persona's own ~/.grok/config.toml, which grok writes as you change settings there,
    | # wins over the same key here. What grok records for itself ([privacy], the [marketplace] install flags,
    | # [cli] installer) stays out of this file: grok keeps it in each persona's own.
    | 
    | [[marketplace.sources]]
    | name = "xAI Official"
    | git = "https://github.com/xai-org/plugin-marketplace.git"
    | 
    | [ui]
    | max_thoughts_width = 120
    | theme = "grokday"
    | fork_secondary_model = "grok-4.6"
    | yolo = false
    | compact_mode = false
    | permission_mode = "always-approve"   # grok runs its tools without asking
    | 
    | [models]
    | default = "grok-4.7"
    | default_reasoning_effort = "xhigh"
    | 
    | # Claude Code's files, in a project's .claude/ and in ~/.claude, are for the Claude personas: grok reads none
    | # it has a switch for (CLAUDE.md and rules in .claude, skills and commands, hooks, ~/.claude.json's MCP
    | # servers). Grok 1.0.67 has none for the rest, so it still reads the permissions in .claude/settings.json and
    | # settings.local.json, the subagents in .claude/agents and the plugins in .claude/plugins; and a top-level
    | # CLAUDE.md, whatever these say. `grok inspect` in a persona shows what it took.
    | [compat.claude]
    | rules = false
    | skills = false
    | agents = false
    | mcps = false
    | hooks = false
    | sessions = false
    | 
    | [tools]
    | respect_gitignore = false
    | 
    | [features]
    | subagent_model_inheritance = false
    | 
    | 
    | # The Nexus notes, /etc/grok/rules/00-nexus.md: nexus --config adds this table.
    | [paths]
    | extra_rule_dirs = ["/etc/grok/rules"]
