#CLAUDE_TAG=
#GROK_TAG=

# Build pins. `nexus <persona> upgrade` sets CLAUDE_VERSION or GROK_VERSION to the newest release and
# rebuilds. DDEV_VERSION only matters when building ddev-context images here; CLAUDE_VERSION and
# GROK_VERSION are also the image tags (unless the *_TAG is set), so on a pulling VM they pick the tag
# that is pulled and run.
# Claude Code (Anthropic's native installer): an exact X.Y.Z, or a channel, latest | stable.
# A pin is reproducible, and bumping it re-runs only the Claude installer stage. A
# channel is resolved once and then cached: `nexus <persona> build --no-cache` to move to a newer
# release. Pins below 2.1.118 ignore the image's lock on `claude update`. Current versions:
#   curl -fsSL https://downloads.claude.ai/claude-code-releases/latest   (or …/stable)
# Grok Build (xAI's installer): an exact X.Y.Z, or latest (the newest on its stable channel), cached once
# built like a Claude channel. Current version:  curl -fsSL https://x.ai/cli/stable
GROK_VERSION=latest
# APM (Agent Package Manager, in both flavours), from PyPI's apm-cli: an exact X.Y.Z, or latest.
# As with Claude, a pin bump re-runs only the APM stage and latest is cached once built. Current version:
#   curl -fsSL https://pypi.org/pypi/apm-cli/json | jq -r .info.version
APM_VERSION=latest
# ddev context only: pin ddev to `ddev version` on this VM (nexus --config does; the leading v is optional),
# or the router gets recreated on every switch.
DDEV_VERSION=

# Per-container ceiling. Six parallel Claude sessions sit comfortably in this.
# AGENT_CPUS must not exceed the VM's vCPU count (nproc), or the container won't start.
AGENT_CPUS=3
AGENT_MEMORY=8g

# its bearer token (this file is made private, mode 600, once it holds one); the CA its certificate is signed by,
